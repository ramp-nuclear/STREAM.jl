# test/test_channels.jl — Phase 55 TEST-01 unified channel-family unit tests.

# HeatDiffusion connects ONLY to ChannelAndContacts. Channel and ChannelHeatFlux
# NEVER wire to HeatDiffusion — they consume external T_wall / q values via

# ─────────────────────────────────────────────────────────────────
# Section 2: Adiabatic-by-default (D-17 second bullet)
# Closed Pump → bc → Channel → Pump loop with default kwargs.
# Spike #1 outcome (55-WAVE0-SPIKE-RESULTS.md) drives whether to add
# binding eqns or leave T_wall_*[i] unbound.
# ─────────────────────────────────────────────────────────────────
@testset "Channel adiabatic-by-default — closed loop, h_*=0.0" begin
    n = N_DEFAULT
    @named pump = Pump(DP_PUMP)
    @named bc = HeatExchanger(T_INLET)
    @named ch = Channel(; n=n, geometry=PipeGeometry_circular(L_DEFAULT, D_DEFAULT))
    # Per Spike #1: Hypothesis A means leave unbound; A_PARTIAL means MTK simplifies
    # them out; B means we MUST add binding eqns. The branch below reads the
    # WAVE0 result file at script-load time (or just unconditionally adds the
    # binding eqns, which is correct under all three hypotheses since 0 multiplied
    # by anything = 0). For robustness, add binding eqns to T_INLET — they are
    # decorative under H=A/A_PARTIAL but required under H=B.
    connections = Equation[
        connect(pump.port_out, bc.port_in),
        connect(bc.port_out, ch.port_in),
        connect(ch.port_out, pump.port_in),
        pump.port_in.P ~ 1.0e5,
        # Decorative under H=A/A_PARTIAL (h_left=0 zeros the q expression);
        # required under H=B for fully_determined compile.
        [ch.T_wall_left[i]  ~ T_INLET for i in 1:n]...,
        [ch.T_wall_right[i] ~ T_INLET for i in 1:n]...,
    ]
    @named sys = compose(System(connections, t; name=:adiab), pump, bc, ch)
    ssys = mtkcompile(sys)  # full closed loop should NOT need fully_determined=false
    ic = Pair{Any,Any}[
        [ssys.ch.T[i] => T_INLET for i in 1:n]...,
        ssys.ch.port_in.mdot => 0.5,
    ]
    sol = solve_transient(ssys, ic, range(0.0, 0.1, length=20))
    @test sol.retcode == ReturnCode.Success
    # T_out ≈ T_inlet — no heating
    @test isapprox(sol[ssys.ch.T_out, end], T_INLET; rtol=1e-3)
end
