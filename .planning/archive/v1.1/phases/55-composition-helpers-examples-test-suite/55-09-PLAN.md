```julia
"""
    build_loop_lof_bypass(; n=10, L_ch=1.0, D_ch=0.01, T_wall=373.15, T_inlet=313.15,
                            h_wall=5000.0, L_over_A=1.75e5, g_acc=9.80665,
                            R_ext=1.0e6, dt_ramp=5.0) -> ODESystem

Build a loss-of-flow validation loop with bypass topology. Heated leg uses
`ChannelAndContacts` pinned via `WallTemperature` source — Phase 55 Spike A topology.

Topology (4-node parallel network):
- Node A (top): ine output, ch input, flapper input (3-way junction)
- Node B (bottom): ch output, ret input (2-way)
- Node C (top): ret output, flapper output, ext_res input (3-way junction)
- D series branch: ext_res -> hx -> pump -> ine

Gravity signs:
- ch (CAC, A->B, nominally downward): g = -g_acc
- ret (Channel, B->C, nominally upward): g = +g_acc

Physics: Pump coasts to 0 dP. Inertia carries momentum; ch flow decays. Flapper
opens when pump branch mdot drops below internal threshold. After Flapper opens,
flow redistributes: ch flow reverses (upward NC driven by buoyancy).

# Returns
Compiled `ODESystem` (via `mtkcompile(sys)`).
"""
#! format: off
function build_loop_lof_bypass(;
    n::Int    = 10,
    L_ch      = 1.0,
    D_ch      = 0.01,
    T_wall    = 373.15,
    T_inlet   = 313.15,
    h_wall    = 5000.0,
    L_over_A  = 1.75e5,
    g_acc     = 9.80665,
    R_ext     = 1.0e6,
    dt_ramp   = 5.0,
)
#! format: on
    geom = PipeGeometry_circular(L_ch, D_ch)
    rd_ch = regime_dependent(;
        htc_laminar=constant_Nusselt(; Nu=8.235),
        htc_turbulent=dittus_boelter,
        friction_laminar=laminar_friction(1.0),
        friction_turbulent=blasius_friction,
        htc_natural=elenbaas_htc(; b=D_ch, L=L_ch, Dh=D_ch, g=g_acc),
        Dh=D_ch,
        g=g_acc,
    )

    @named pump = Pump(0.0)
    @named ine  = Inertia(L_over_A)
    @named hx   = HeatExchanger(T_inlet)
    @named ch   = ChannelAndContacts(; n=n, geometry=geom, g=(-g_acc),
                                     htc_correlation=rd_ch.htc,
                                     friction_correlation=rd_ch.friction)
    @named ret  = Channel(; n=n, geometry=geom, g=g_acc)
    @named flapper = Flapper(; dt=dt_ramp)
    @named ext_res = Resistor(R_ext)
    @named ch_wt = WallTemperature(; n=n, T_wall=T_wall)

    connections = Equation[
        connect(ext_res.port_out, hx.port_in),
        connect(hx.port_out, pump.port_in),
        connect(pump.port_out, ine.port_in),
        connect(ine.port_out, ch.port_in, flapper.port_in),
        connect(ch.port_out, ret.port_in),
        connect(ret.port_out, flapper.port_out, ext_res.port_in),
        pump.port_in.P ~ 1.0e5,
        flapper.ref_mdot ~ ine.port_in.mdot,
        # Heated leg: pin CAC's left thermal port T to WallTemperature source per cell.
        # CAC's right thermal port dangles → adiabatic via ThermalPort Flow rule.
        [ch.thermal_left[i].T ~ ch_wt.T_wall_out[i] for i in 1:n]...,
        # Return leg: ret is the new Channel — adiabatic via h_left=h_right=0 default.
        # Bind T_wall_*[i] to T_inlet (decorative under H=A; required under H=B from Spike #1).
        [ret.T_wall_left[i]  ~ T_inlet for i in 1:n]...,
        [ret.T_wall_right[i] ~ T_inlet for i in 1:n]...,
    ]

    @named sys = compose(System(connections, t; name=:sys),
                          pump, ine, hx, ch, ret, flapper, ext_res, ch_wt)
    t_compile = @elapsed ssys = mtkcompile(sys)
    n_eq = length(equations(ssys))
    n_uk = length(unknowns(ssys))
    @info "build_loop_lof_bypass compile time: $(round(t_compile; digits=2))s" n_equations = n_eq n_unknowns = n_uk

    return ssys
end
```

```julia
"""
    build_loop_lof_bypass(; n=10, L_ch=1.0, D_ch=0.01, T_inlet=313.15,
                            power_W=1.0e3, fuel_nx=2, fuel_Lx=0.005,
                            L_over_A=1.75e5, g_acc=9.80665,
                            R_ext=1.0e6, dt_ramp=5.0) -> ODESystem

Build a loss-of-flow validation loop with bypass topology. Heated leg uses
`ChannelAndContacts + HeatDiffusion` plate via `one_sided_connection` — Phase 55
Spike B topology (real fuel-plate physics).

[Topology + gravity discussion same as Spike A; only the heated-leg implementation differs.]

# Returns
Compiled `ODESystem` (via `mtkcompile(sys)`).
"""
#! format: off
function build_loop_lof_bypass(;
    n::Int    = 10,
    L_ch      = 1.0,
    D_ch      = 0.01,
    T_inlet   = 313.15,
    power_W   = 1.0e3,
    fuel_nx   = 2,
    fuel_Lx   = 0.005,
    L_over_A  = 1.75e5,
    g_acc     = 9.80665,
    R_ext     = 1.0e6,
    dt_ramp   = 5.0,
)
#! format: on
    geom = PipeGeometry_circular(L_ch, D_ch)
    rd_ch = regime_dependent(;
        htc_laminar=constant_Nusselt(; Nu=8.235),
        htc_turbulent=dittus_boelter,
        friction_laminar=laminar_friction(1.0),
        friction_turbulent=blasius_friction,
        htc_natural=elenbaas_htc(; b=D_ch, L=L_ch, Dh=D_ch, g=g_acc),
        Dh=D_ch,
        g=g_acc,
    )

    @named pump = Pump(0.0)
    @named ine  = Inertia(L_over_A)
    @named hx   = HeatExchanger(T_inlet)
    @named ch   = ChannelAndContacts(; n=n, geometry=geom, g=(-g_acc),
                                     htc_correlation=rd_ch.htc,
                                     friction_correlation=rd_ch.friction)
    @named ret  = Channel(; n=n, geometry=geom, g=g_acc)
    @named flapper = Flapper(; dt=dt_ramp)
    @named ext_res = Resistor(R_ext)

    ps = fill(1.0 / (n * fuel_nx), n, fuel_nx)
    @named fuel = HeatDiffusion(; nz=n, nx=fuel_nx, Lz=L_ch, Lx=fuel_Lx,
                                 y=0.07, rho_s=19300.0, cp_s=116.0, k_s=174.0,
                                 power_shape=ps)
    heated = one_sided_connection(ch, fuel; side=:left, name=:heated)

    connections = Equation[
        connect(ext_res.port_out, hx.port_in),
        connect(hx.port_out, pump.port_in),
        connect(pump.port_out, ine.port_in),
        connect(ine.port_out, heated.channel.port_in, flapper.port_in),
        connect(heated.channel.port_out, ret.port_in),
        connect(ret.port_out, flapper.port_out, ext_res.port_in),
        pump.port_in.P ~ 1.0e5,
        flapper.ref_mdot ~ ine.port_in.mdot,
        heated.fuel.power ~ power_W,
        # Return leg: ret is the new Channel — adiabatic via h_*=0 default.
        [ret.T_wall_left[i]  ~ T_inlet for i in 1:n]...,
        [ret.T_wall_right[i] ~ T_inlet for i in 1:n]...,
    ]

    @named sys = compose_systems(heated, pump, ine, hx, ret, flapper, ext_res;
                                 connections=connections, name=:sys)
    t_compile = @elapsed ssys = mtkcompile(sys)
    n_eq = length(equations(ssys))
    n_uk = length(unknowns(ssys))
    @info "build_loop_lof_bypass compile time: $(round(t_compile; digits=2))s" n_equations = n_eq n_unknowns = n_uk

    return ssys
end
```
