# ========================================================================
# G3 — Single-cell forward/reverse flow mirror (rtol=1e-12, fallback 1e-9)
# ========================================================================
# ROADMAP success criterion #4: same ifelse(mdot >= 0, ...) selects upstream T,
# and (deterministically) cp(T_up). On a single cell with matched q profile and
# opposite mdot signs, dT_fwd ≈ -dT_rev to analytical precision.
@testset "G3: Single-cell forward/reverse mirror" begin
    n = 1
    geom = PipeGeometry_circular(0.1, 0.01)  # short tube, single cell
    Q = 1000.0  # W per cell — modest so tolerance is not solver-limited
    q_left_vals  = [Q]
    q_right_vals = [0.0]
    T_in = 320.0  # K — set via HeatExchanger

    function _build_loop(mdot0)
        @named pump = Pump(; mdot0=mdot0)
        @named hex  = HeatExchanger(T_in)
        @named stub = _StubChannelCore(; n=n, geometry=geom,
                                        q_left_vals=q_left_vals, q_right_vals=q_right_vals)
        eqs = Equation[
            connect(pump.port_out, hex.port_in),
            connect(hex.port_out,  stub.port_in),
            connect(stub.port_out, pump.port_in),
            pump.port_in.P ~ 1.0e5,
        ]
        nm = mdot0 > 0 ? :g3_loop_fwd : :g3_loop_rev
        @named sys = compose(System(eqs, t; name=nm), pump, hex, stub)
        ssys = mtkcompile(sys)
        sol = solve_steady(ssys, []; abstol=1e-12, reltol=1e-12)
        return ssys, sol
    end

    ssys_fwd, sol_fwd = _build_loop(+0.1)
    ssys_rev, sol_rev = _build_loop(-0.1)

    @test sol_fwd.retcode == ReturnCode.Success
    @test sol_rev.retcode == ReturnCode.Success

    if sol_fwd.retcode == ReturnCode.Success && sol_rev.retcode == ReturnCode.Success
        # Forward: T_in flows in via port_in, exits via port_out. T_out (= T[n]) is the outlet.
        T_out_fwd = sol_fwd[ssys_fwd.stub.T_out]
        # Reverse: flow direction is port_out -> port_in. T_in_rev is applied at port_out
        # (HeatExchanger upstream of pump.port_in becomes downstream under reverse mdot).
        # The "outlet" in the reverse case is T[1] (port_in side becomes downstream).
        # See RESEARCH.md line 644-648 for the indexing rationale.
        T_out_rev = sol_rev[ssys_rev.stub.T[1]]

        dT_fwd = T_out_fwd - T_in
        dT_rev = T_out_rev - T_in

        # Strict mirror — rtol=1e-12 with explicit solver tolerances.
        # If this is flaky on this machine (KINSOL on the closed loop has
        # bounded precision), fall back to rtol=1e-9 with the comment below.
        try
            @test isapprox(dT_fwd, -dT_rev; rtol=1e-12)
        catch err
            # Tolerance fallback per VALIDATION.md G3 note.
            @warn "G3 rtol=1e-12 failed; relaxing to rtol=1e-9. dT_fwd=$dT_fwd dT_rev=$dT_rev"
            @test isapprox(dT_fwd, -dT_rev; rtol=1e-9)
        end
    end
end

# ========================================================================
# G4 — Branch-coverage matrix (CORE-05; ROADMAP success criterion #5)
# ========================================================================
# Every code path in _channel_core must be reachable by at least one stub configuration.
# Branches in the new core (per Plan 02 output):
#   B1: forward-flow boundary face (i=1, port_in.mdot >= 0) -- selects T_inlet_fwd
#   B2: reverse-flow boundary face (i=n, port_in.mdot < 0)   -- selects T_inlet_rev
#   B3: interior face (1 < i < n)                            -- selects T[i±1]
#   B4: adiabatic                                            -- q_left=q_right=0
