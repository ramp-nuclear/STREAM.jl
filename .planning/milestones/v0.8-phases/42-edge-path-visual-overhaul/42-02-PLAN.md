**1. Fix reconstructInstanceCounters in projectIO.ts (lines 138-160):**
Replace the entire function body with the componentId-based implementation per D-09:
```typescript
export function reconstructInstanceCounters(
  nodes: Node[],
): Record<string, number> {
  const counters: Record<string, number> = {};

  for (const node of nodes) {
    const data = node.data as { componentId?: string; instanceName?: unknown };
    if (!data?.componentId || typeof data.instanceName !== "string") continue;

    const key = data.componentId.toLowerCase();
    const pattern = new RegExp(`^${key}_(\\d+)$`);
    const match = pattern.exec(data.instanceName);
    if (!match) continue;

    const num = parseInt(match[1], 10);
    if (counters[key] === undefined || num > counters[key]) {
      counters[key] = num;
    }
  }

  return counters;
}
```

Key changes from the old version:
- Uses `data.componentId.toLowerCase()` as the key instead of extracting prefix from instanceName via generic regex
- Only matches `instanceName` against `^<componentId_lower>_(\d+)$` pattern
- Custom-renamed nodes (e.g., `my_special_pump`) are correctly ignored because they don't match `^pump_(\d+)$`
- Nodes like `my_custom_3` no longer pollute the counter for key `my_custom`
