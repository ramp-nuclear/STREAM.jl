    ```julia
    # helpers.jl — QoL and composition helpers for STREAM.jl
    #
    # QoL helpers (Phase 15 Plan 01): port, check_gravity_mismatch
    # Composition helpers (Phase 15 Plan 02): symmetric_plate, plate,
    #                                          one_sided_connection, compose_systems

    # ----------------------------------------------------------------
    # port — QOL-03
    # Wraps getproperty(sys, Symbol(face, i)) — the only correct MTK
    # syntax for indexed port array access in connect() calls.
    #
    # Example:
    #   connect(port(cac, :thermal_left, 2), port(fuel, :thermal_right, 2))
    # ----------------------------------------------------------------
    port(sys, face::Symbol, i::Int) = getproperty(sys, Symbol(face, i))

    # ----------------------------------------------------------------
    # check_gravity_mismatch — QOL-02
    # Checks whether gravity pressure contributions sum to zero at zero flow.
    # Returns :ok if balanced (|residual| < 1.0 Pa), :mismatch otherwise.
    #
    # Algorithm: collect all equations from the system (using full_equations
    # for pre-compiled systems, equations() for compiled), filter those containing
    # the gravity parameter g_acc, substitute all mdot symbols to 0.0, and sum
    # the resulting residuals.
    #
    # Works on both pre- and post-mtkcompile systems.
    # ----------------------------------------------------------------
    function check_gravity_mismatch(sys::ModelingToolkit.AbstractSystem)
        # Try to get flattened equations — full_equations works pre-compile
        all_eqs = try
            ModelingToolkit.full_equations(sys)
        catch
            ModelingToolkit.equations(sys)
        end

        # Collect all unknown variables (includes mdot from compiled systems)
        all_unknowns = try
            ModelingToolkit.unknowns(sys)
        catch
            Num[]
        end

        # Find mdot variables to substitute to zero
        mdot_subs = Dict{Any,Float64}()
        for v in all_unknowns
            if occursin("mdot", string(v))
                mdot_subs[v] = 0.0
            end
        end

        # Also substitute any mdot parameters
        all_pars = try
            ModelingToolkit.parameters(sys)
        catch
            Num[]
        end
        for p in all_pars
            if occursin("mdot", string(p))
                mdot_subs[p] = 0.0
            end
        end

        # Filter and evaluate pressure/gravity equations
        gravity_residuals = Float64[]
        for eq in all_eqs
            eq_str = string(eq)
            # Look for equations involving g_acc (gravity parameter name in Gravity component
            # and ChannelAndContacts dP equation)
            if occursin("g_acc", eq_str) || occursin("dP", string(eq.lhs))
                # Substitute mdot=0 into the residual (lhs - rhs)
                residual_expr = Symbolics.substitute(eq.lhs - eq.rhs, mdot_subs)
                try
                    val = Float64(Symbolics.unwrap(residual_expr))
                    push!(gravity_residuals, val)
                catch
                    # Cannot reduce to Float64 (T unknowns remain) — skip
                end
            end
        end

        if isempty(gravity_residuals)
            # No gravity equations found or none evaluable — assume ok
            return :ok
        end

        # Sum of gravity residuals: in a balanced loop, pressure drops cancel
        total_residual = sum(gravity_residuals)
        atol = 1.0  # Pa — well below any real pressure scale
        if abs(total_residual) > atol
            @warn "check_gravity_mismatch: gravity pressure residual at zero flow = $(round(total_residual; digits=2)) Pa — loop gravity terms may be unbalanced"
            return :mismatch
        end
        return :ok
    end
    ```
