```julia
function _diffusion_eqs(eqs::Vector{Equation};
    T, thermal_left, thermal_right,
    nz, nx, k_s, rho_s, cp_s, dx, dz, y, power, power_shape, Dt)
    # v0.4: add dz, kz arguments for xz-diffusion (DIFF-01)
```

The function appends equations for every axial cell i in 1:nz:

**1. Left boundary Q_flow (i in 1:nz) — LOCKED formula:**
```julia
push!(eqs, thermal_left[i].Q_flow ~
    k_s * (y * dz) * (T[i, 1] - thermal_left[i].T) / (dx / 2))
```

**2. Right boundary Q_flow (i in 1:nz) — LOCKED formula:**
```julia
push!(eqs, thermal_right[i].Q_flow ~
    k_s * (y * dz) * (thermal_right[i].T - T[i, nx]) / (dx / 2))
```

**3. Temperature ODEs — Option B (ODE for ALL nz*nx cells, including boundary cells j=1 and j=nx):**

For each i in 1:nz, for each j in 1:nx:
- `q_vol = power * power_shape[i, j] / (rho_s * cp_s * y * dz * dx)` (volumetric heat source)
- j == 1 (left boundary cell, uses thermal_left[i].T as virtual left neighbor at distance dx/2):
  ```julia
  push!(eqs, Dt(T[i, 1]) ~
      k_s / (rho_s * cp_s) *
      (T[i, 2] - T[i, 1]) / dx^2 * 2 -   # right flux: (T[i,2]-T[i,1])/dx / (dx/2+dx/2)
      k_s / (rho_s * cp_s) *
      (T[i, 1] - thermal_left[i].T) / ((dx/2)^2) * ... + q_vol)
  ```
  Use a consistent half-cell scheme: the left boundary cell center is at x=dx/2 from the left face. The flux to the right is `k_s*(T[i,2]-T[i,1])/dx` (full dx between centers). The flux to the left face (boundary) is `k_s*(T[i,1]-thermal_left[i].T)/(dx/2)`. Net: `(right_flux - left_flux) / (y*dz*dx)`. The ODE for j=1:
  ```julia
  Dt(T[i, 1]) ~ k_s / (rho_s * cp_s * dx^2) *
      (T[i, 2] - T[i, 1] - 2*(T[i, 1] - thermal_left[i].T)) + q_vol
  ```
  Wait — use the explicit flux form for clarity:
  ```julia
  Dt(T[i, 1]) ~ (k_s * (T[i, 2] - T[i, 1]) / dx
                - k_s * (T[i, 1] - thermal_left[i].T) / (dx / 2)) /
                (rho_s * cp_s * dx) + q_vol
  ```

- 2 <= j <= nx-1 (interior cells — LOCKED formula from CONTEXT.md):
  ```julia
  push!(eqs, Dt(T[i, j]) ~
      k_s * (T[i, j+1] - 2*T[i, j] + T[i, j-1]) / (dx^2 * rho_s * cp_s)
      + q_vol)
  ```

- j == nx (right boundary cell, uses thermal_right[i].T as virtual right neighbor at distance dx/2):
  ```julia
  Dt(T[i, nx]) ~ (k_s * (thermal_right[i].T - T[i, nx]) / (dx / 2)
                 - k_s * (T[i, nx] - T[i, nx-1]) / dx) /
                 (rho_s * cp_s * dx) + q_vol
  ```

No z-diffusion terms are written at all — adiabatic top/bottom by omission (NOT by explicit zero-flux equations).

The `power_shape` argument is indexed as `power_shape[i, j]` — row i (axial), column j (lateral). No internal normalization.

NOTE: if nx == 1, only the boundary cells exist — handle gracefully (j=1 and j=nx are the same cell; use averaged stencil or handle as a degenerate case. For v0.3, require nx >= 2 with an assertion or simply let the j=1/j=nx logic cover it without interior loop).

**In `src/components.jl`**, immediately after `_diffusion_eqs`, add the `HeatDiffusion` constructor:

```julia
# HeatDiffusion: 2D finite-difference solid fuel plate with x-direction diffusion only (v0.3).
# State T(t)[1:nz, 1:nx]: row i = axial cell (index 1 = inlet/top), col j = lateral cell.
# thermal_left[1:nz], thermal_right[1:nz]: ThermalPort arrays for coupling to coolant channels.
# Material properties (rho_s, cp_s, k_s) are plain Float64 — not MTK parameters (v0.3).
# power: MTK @parameters (tunable via remake()), total watts into the plate.
# power_shape[nz, nx]: user-supplied spatial distribution — NOT normalized internally.
# Top/bottom boundaries are adiabatic by omission of z-diffusion equations.
function HeatDiffusion(; name,
                         nz::Int, nx::Int,
                         Lz, Lx, y,
                         rho_s, cp_s, k_s,
                         power_shape,
                         power  = 1e6,
                         T0     = 600.0)
    Dt = Differential(t)
    dx = Lx / nx
    dz = Lz / nz

    pars = @parameters begin
        power = power
    end

    vars = @variables begin
        (T(t))[1:nz, 1:nx] = fill(T0, nz, nx)
    end

    thermal_left  = [ThermalPort(name=Symbol(:thermal_left, i))  for i in 1:nz]
    thermal_right = [ThermalPort(name=Symbol(:thermal_right, i)) for i in 1:nz]

    eqs = Equation[]
    _diffusion_eqs(eqs;
        T            = T,
        thermal_left  = thermal_left,
        thermal_right = thermal_right,
        nz = nz, nx = nx,
        k_s = k_s, rho_s = rho_s, cp_s = cp_s,
        dx = dx, dz = dz, y = y,
        power        = only(pars),
        power_shape  = power_shape,
        Dt           = Dt)

    all_vars = vec(collect(T))

    compose(System(eqs, t, all_vars, pars; name=name),
            thermal_left..., thermal_right...)
end
```

Key implementation notes:
- `only(pars)` extracts the single MTK parameter symbol (equivalent to `pars[1]`). Alternatively use `pars[1]` directly — both are fine.
- `vec(collect(T))` flattens the 2D array state to a 1D variable list for `System(...)`. This is the established pattern for 2D MTK state (see RESEARCH.md Pitfall 2).
- `compose(System(...), thermal_left..., thermal_right...)` — no `port_in`/`port_out` (solid component, no flow ports). Follow the splat ordering from ChannelAndContacts.
- No `@named` macro needed inside the function for `thermal_left`/`thermal_right` — they are created with explicit `name=Symbol(...)` kwargs.
