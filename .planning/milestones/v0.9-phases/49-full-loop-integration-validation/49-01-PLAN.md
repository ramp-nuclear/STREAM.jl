The `temp_worth` and `ref_temp` kwargs accept `Dict{Symbol,Any}` where keys are `:cac` or `:fuel`. Inside the function, resolve to scoped refs:

```julia
# Resolve Symbol keys to scoped component refs
function _resolve_tw(d, rods)
    isnothing(d) && return nothing
    resolved = Dict{Any,Any}()
    for (k, v) in d
        comp = k == :cac ? rods.cac : k == :fuel ? rods.fuel : error("Unknown component key: $k")
        resolved[comp] = v
    end
    return resolved
end

tw = _resolve_tw(temp_worth, rods)
rt = _resolve_tw(ref_temp, rods)
```

Then construct PK:
```julia
@named pk = PointKinetics(ctrl; rho_val=rho_val, temp_worth=tw, ref_temp=rt)
```

**Stage 3 -- Connections and compose (per D-01, D-02):**
```julia
fb_eqs    = isnothing(tw) ? Equation[] : connect_temperature_feedback(pk, [rods.cac, rods.fuel])

@named pump = Pump(dP_pump)
@named bc   = HeatExchanger(T_inlet)

all_connections = [
    connect(pump.port_out, bc.port_in),
    connect(bc.port_out, rods.cac.port_in),
```

**LOOP-03 test (step reactivity with negative feedback):**
```julia
@testset "LOOP-03: step reactivity with temperature feedback" begin
    P0        = 1.0
    t_step    = 0.5
    delta_rho = 0.0005
    alpha     = -0.01
    T_inlet   = 293.15

    step_fn = t -> (t >= t_step ? delta_rho : 0.0)
    ctrl = ReactivityController(step_fn)

    ssys, ic = build_loop_pk(ctrl;
        P0=P0, power_scale=1e4,
        temp_worth = Dict(:cac => fill(alpha, 7)),
        ref_temp   = Dict(:cac => fill(T_inlet, 7)),
    )

    t_arr = range(0.0, 5.0; length=500)
    sol = solve_transient(ssys, ic, t_arr; tstops=[t_step], maxiters=1_000_000)

    P_trace = sol[ssys.pk.P, :]
    P_max   = maximum(P_trace)

    @test P_max > P0                    # power rises after step
    @test P_trace[end] < P_max          # feedback damps the excursion
    @test all(isfinite, P_trace)        # no NaN/Inf
end
```

**LOOP-04 test (SCRAM-in-loop, following SCRAM-02 pattern from test_point_kinetics.jl):**

```julia
@testset "LOOP-04: SCRAM terminates coupled loop" begin
    P0        = 1.0
    plimit    = 1.2
    t_step    = 0.5
    delta_rho = 0.005   # large enough to exceed plimit quickly
    alpha     = -0.01
    T_inlet   = 293.15

    scram_ir = (state, t_state, t) -> state == :SCRAM ? -0.05 : (t >= t_step ? delta_rho : 0.0)
    ctrl = ReactivityController(scram_ir;
        initial_state = :NORMAL,
        state_machine = SCRAM_at_power(plimit),
        abort_states  = Set([:SCRAM]))

    ssys, ic = build_loop_pk(ctrl;
        P0=P0, power_scale=1e4,
        temp_worth = Dict(:cac => fill(alpha, 7)),
        ref_temp   = Dict(:cac => fill(T_inlet, 7)),
    )

    cb = scram_callback(ssys, ssys.pk.P, ctrl)

    t_arr = range(0.0, 10.0; length=1000)
    sol = solve_transient(ssys, ic, t_arr; tstops=[t_step], callbacks=cb, maxiters=1_000_000)

    @test sol.t[end] < 10.0            # terminated early
    @test ctrl.state == :SCRAM         # state machine transitioned
    @test any(entry -> entry[1] == :SCRAM, ctrl.log)  # SCRAM logged
end
```
