**VAL-PK-01: Steady-state linear temperature rise (matches Python STREAM lines 201-267):**
```julia
@testset "VAL-PK-01: steady-state coolant temperature rises linearly" begin
    n = 7
    T_inlet = 293.15
    ctrl = ReactivityController()
    ssys, ic = build_loop_pk(ctrl; n=n, T_inlet=T_inlet, P0=1.0, power_scale=1e4)

    # Attempt steady-state solve; fall back to long transient if KINSOL fails
    local T_cool
    try
        sol = solve_steady(ssys, ic)
        T_cool = [sol[ssys.rods.cac.T[i]] for i in 1:n]
    catch e
        # Fallback: run transient to steady state
        t_arr = range(0.0, 100.0; length=500)
        sol = solve_transient(ssys, ic, t_arr; maxiters=1_000_000)
        T_cool = [sol[ssys.rods.cac.T[i], end] for i in 1:n]
    end

    dT = diff(T_cool)
    ddT = diff(dT)

    @test all(dT .> 0)                                   # strictly rising
    @test isapprox(ddT, zeros(length(ddT)); atol=0.5)    # approximately linear
end
```

**VAL-PK-02a: Negative fuel feedback suppresses power (matches Python STREAM lines 352-387):**

The Python reference uses `ref_temp` at the boundary condition temperature and a negative alpha on the fuel. Since fuel temperature exceeds the reference, negative feedback drives power down.

```julia
@testset "VAL-PK-02a: negative fuel feedback suppresses power to near zero" begin
    n = 7; nz = 7; nx = 2
    T_inlet  = 293.15
    T_fuel_ref = 600.0    # HeatDiffusion initial T -- becomes ref_temp
    alpha_neg  = -0.1     # strong negative feedback (same magnitude as Python STREAM)

    ctrl = ReactivityController()
    ssys, ic = build_loop_pk(ctrl;
        n=n, nz=nz, nx=nx, T_inlet=T_inlet,
        P0=1.0, power_scale=1e4,
        temp_worth = Dict(:fuel => fill(alpha_neg, nz * nx)),
        ref_temp   = Dict(:fuel => fill(T_fuel_ref, nz * nx)),
    )

    # Override PK ICs to large values (per Pitfall 4: helps KINSOL find P~0 solution)
    ic_high = copy(ic)
    for (idx, pair) in enumerate(ic_high)
        if pair.first === ssys.pk.P
            ic_high[idx] = ssys.pk.P => 1e3
        end
        for k in 1:6
            sym = getproperty(ssys.pk, Symbol(:C_, k))
            if pair.first === sym
                ic_high[idx] = sym => 1e3
            end
        end
    end

    local P_final
    try
        sol = solve_steady(ssys, ic_high)
        P_final = sol[ssys.pk.P]
    catch e
        # Fallback: long transient
        t_arr = range(0.0, 200.0; length=500)
        sol = solve_transient(ssys, ic_high, t_arr; maxiters=1_000_000)
        P_final = sol[ssys.pk.P, end]
    end

    @test abs(P_final) < 1e-3    # power driven to near zero by negative feedback
end
```

**VAL-PK-03: Reactivity observable accessible and correct:**
```julia
@testset "VAL-PK-03: reactivity observable accessible and correct at steady state" begin
    n = 7
    T_inlet = 293.15
    alpha   = -0.005
    ctrl = ReactivityController()

    ssys, ic = build_loop_pk(ctrl;
        n=n, T_inlet=T_inlet,
        P0=1.0, power_scale=1e4,
        temp_worth = Dict(:cac => fill(alpha, n)),
        ref_temp   = Dict(:cac => fill(T_inlet, n)),
    )

    t_arr = range(0.0, 50.0; length=200)
    sol = solve_transient(ssys, ic, t_arr; maxiters=1_000_000)

    # Reactivity observable must be accessible
    rho_trace = sol[ssys.pk.reactivity, :]
    @test rho_trace isa AbstractVector
    @test length(rho_trace) > 1
    @test all(isfinite, rho_trace)

    # At steady state (final time), reactivity should be near zero
    # (steady state means dP/dt=0, which requires net reactivity ~ 0)
    @test abs(rho_trace[end]) < 0.01
end
```
