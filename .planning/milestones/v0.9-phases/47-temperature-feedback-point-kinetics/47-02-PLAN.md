    @testset "TF-07: strong negative feedback bounds power (analytical)" begin
        # Mirror Python STREAM test_integrations.py:352-428 (fuel feedback & coolant feedback).
        # Setup: inject a positive step reactivity via rho_c_fn; attach strong negative
        # temperature feedback on the channel. Expect:
        #   (1) power rises initially after the step,
        #   (2) power peaks at some t_peak > t_step,
        #   (3) power does NOT diverge — max(P) / P0 is bounded by a physical limit,
        #   (4) at late t, power drops back below the prompt-jump level because the
        #       negative feedback overcomes the inserted reactivity.

        n = 3
        # ... build @named cac = ChannelAndContacts(..., n=n, ...) with inlet T_in=293.0
        # ... build @named fuel = HeatDiffusion(..., nz=n, nx=1, ...)

        t_step   = 0.1          # reactivity insertion time [s]
        delta_rho = 0.0005      # step reactivity (well below beta_total=0.006502 to stay sub-prompt)
        alpha_strong = -0.01    # strong negative alpha [Δk/k per K] — stabilizing
        Tref = 293.0

        # Step reactivity callable: 0 before t_step, delta_rho after.
        step_ctrl = (t) -> (t >= t_step ? delta_rho : 0.0)
        # Wrap in ReactivityController-compatible callable if needed — but a plain closure
        # is fine since PointKinetics only requires `rho_c_fn(t)::Float64`.

        @named pk = PointKinetics(step_ctrl; rho_val=0.0,
                                  temp_worth=Dict(cac => fill(alpha_strong, n)),
                                  ref_temp=Dict(cac => fill(Tref, n)))

        rods = symmetric_plate(cac, fuel; name=:rods)
        fb_eqs = connect_temperature_feedback(pk, Dict(cac => fill(alpha_strong, n)))
        full = compose_systems(rods, pk; connections=fb_eqs, name=:core)
        ssys = mtkcompile(full)

        P0 = 1.0
        ic = point_kinetics_steady_state(P0)
        op = Pair{Any,Any}[ ssys.pk.rho_c_fn => step_ctrl,
                            ssys.pk.P => ic.P,
                            ssys.pk.C_1 => ic.C_k[1], ssys.pk.C_2 => ic.C_k[2],
                            ssys.pk.C_3 => ic.C_k[3], ssys.pk.C_4 => ic.C_k[4],
                            ssys.pk.C_5 => ic.C_k[5], ssys.pk.C_6 => ic.C_k[6] ]
        # Add any additional IC entries (inlet T, pressures, etc.) — copy from TF-06 setup.

        sol = solve_transient(ssys, op, (0.0, 2.0))
        P_trace = sol[ssys.pk.P]
        P_max = maximum(P_trace)

        # Analytical expectation:
        # (1) Power rises after step: maximum after t_step must exceed P0.
        # Find index at t_step; use sol.t to locate.
        tvec = sol.t
        idx_step = findfirst(t -> t >= t_step, tvec)
        @test P_max > P0            # power rises (step reactivity was positive)
        @test P_max < 100 * P0      # power is bounded (does NOT diverge — divergence would be >>100x)
        @test all(isfinite, P_trace)

        # (2) Late-time behavior: with strong alpha=-0.01 and any T rise > 0.05 K,
        #     feedback |alpha * dT| >= 0.0005 will cancel the inserted delta_rho.
        #     Therefore net reactivity at t_end should be <= 0 (power returning to or below P0).
        rho_trace = sol[ssys.pk.reactivity]
        @test rho_trace[end] <= delta_rho  # feedback is subtracting from delta_rho
    end
