The Q_wall step change strategy: use a time-dependent function in the connection equation rather than PresetTimeCallback + setp. This is cleaner and avoids the need to declare Q_wall as a @parameters symbol in build_loop (which was already defined with Q_wall as a literal Float64 in the connection equation).

Strategy: add a new exported function `build_loop_transient` that uses a `@parameters Q_wall` symbol so setp can modify it. This keeps build_loop (the steady-state version) unchanged.

Add to solvers.jl:

```julia
# ----------------------------------------------------------------
# build_loop_transient
# Same as build_loop but with Q_wall as a @parameters symbol
# (required for PresetTimeCallback + setp step change).
# Returns (ssys, Q_wall_sym) where Q_wall_sym is the compiled parameter symbol.
# ----------------------------------------------------------------
function build_loop_transient(;
    n::Int   = 10,
    L_ch     = 0.6,
    D_ch     = 0.01,
    A_ch     = 7.85e-5,
    L_fr     = 0.3,
    D_fr     = 0.01,
    A_fr     = 7.85e-5,
    dP_pump  = 3.0e4,
    Q_wall_0 = 1.0e4,
    T_wall   = 373.15,
)
    @named pump = Pump(dP_pump = dP_pump)
    @named fr   = Friction(L = L_fr, D = D_fr, A = A_fr)
    @named ch   = Channel(n = n, L = L_ch, D = D_ch, A = A_ch)

    # Declare Q_wall as a parameter so setp can modify it at runtime
    ps = @parameters Q_wall = Q_wall_0

    connections = [
        connect(pump.outlet, fr.inlet),
        connect(fr.outlet,   ch.inlet),
        connect(ch.outlet,   pump.inlet),
        pump.inlet.P  ~ 1.0e5,
        ch.thermal.Q_flow ~ ps[1],   # Q_wall parameter (not literal)
        ch.thermal.T ~ T_wall,
    ]

    @named sys = compose(
        System(connections, t, [], ps; name = :sys),
        pump, fr, ch
    )

    ssys = mtkcompile(sys)
    return ssys, ps[1]
end

# ----------------------------------------------------------------
# solve_transient
# Simulates the closed loop with a step change in wall heat input.
#
# ssys, Q_wall_sym: from build_loop_transient()
# op:   Vector{Pair} -- initial conditions (same as solve_steady op)
# tspan: (t_start, t_end) in seconds, e.g. (0.0, 60.0)
# Q_wall_final: new Q_wall value after step (W), e.g. 2.0e4
# t_step: time of step change (s), default 10.0
#
# Returns ODESolution. Access time-series:
#   sol[ssys.ch.T_out, :]      -- outlet T at all time points
#   sol.t                       -- time vector
# ----------------------------------------------------------------
function solve_transient(ssys, Q_wall_sym, op, tspan;
                         Q_wall_final,
                         t_step = 10.0)
    prob = ODEProblem(ssys, op, tspan)

    # PresetTimeCallback: fires at t_step, sets Q_wall to Q_wall_final
    Q_setter = setp(ssys, Q_wall_sym)
    step_cb  = PresetTimeCallback(
        [t_step],
        integrator -> Q_setter(integrator, Q_wall_final)
    )

    sol = solve(prob, IDA(); callback = step_cb)
    return sol
end
```
