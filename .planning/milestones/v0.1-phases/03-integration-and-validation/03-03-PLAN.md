```python
#!/usr/bin/env python3
"""
generate_reference.py -- Python STREAM reference value generator

Run this script ONCE to obtain reference values for Julia-STREAM VAL-01 tests:
  cd /home/itay/projects/Julia-STREAM/test && python generate_reference.py

Requires Python STREAM to be installed and importable from ~/projects/STREAM.

UNIT CONVENTION:
  Python STREAM uses CELSIUS for temperature inputs/outputs.
  Julia-STREAM uses KELVIN.
  This script converts all outputs to KELVIN before printing.
  Verification: 313.15 K - 273.15 = 40.0 degC (inlet temperature).
"""

import sys
import os

# Add Python STREAM to path
STREAM_PATH = os.path.expanduser("~/projects/STREAM")
sys.path.insert(0, STREAM_PATH)

# Unit conversion check
T_inlet_K = 313.15
T_inlet_C = 40.0
assert abs(T_inlet_K - 273.15 - T_inlet_C) < 1e-9, "Unit conversion sanity check failed"

# Reference geometry (identical to Julia-STREAM test case)
# Channel: n=10, L=0.6m, D_h=0.01m, A=7.85e-5 m^2 (circular tube, D=1cm)
# Friction: L=0.3m, D=0.01m, A=7.85e-5 m^2
# Pump: dP_pump = 3.0e4 Pa (30 kPa)
# Q_wall = 10,000 W (10 kW uniform)
# T_inlet = 40.0 degC (= 313.15 K)

N_CELLS   = 10
L_CH      = 0.6       # m
D_H       = 0.01      # m
A_CH      = 7.85e-5   # m^2
L_FR      = 0.3       # m
A_FR      = 7.85e-5   # m^2
DP_PUMP   = 3.0e4     # Pa
Q_WALL_W  = 10_000.0  # W (total)

try:
    from stream.substances.light_water import LightWater
    from stream.components.channel import Channel
    from stream.components.pump import Pump
    from stream.components.friction import FrictionFactor
    from stream.aggregator.aggregator import Aggregator

    fluid = LightWater()

    # Instantiate components (Python STREAM uses Celsius)
    ch   = Channel(fluid=fluid, n=N_CELLS, L=L_CH, Dh=D_H, A=A_CH)
    pump = Pump(dP=DP_PUMP)
    fr   = FrictionFactor(fluid=fluid, L=L_FR, D=D_H, A=A_FR)

    # Assemble and solve
    agr = Aggregator([pump, fr, ch])
    agr.set_inlet(T=T_inlet_C, P=1.0e5)

    q_per_cell = Q_WALL_W / N_CELLS
    ch.set_heat([q_per_cell] * N_CELLS)

    sol = agr.solve_steady()

    # Extract results -- Python STREAM returns Celsius
    T_outlet_C  = sol["T_out"]   # adjust key to match actual Python STREAM API
    mdot        = sol["mdot"]    # kg/s -- no unit conversion needed

    # Convert to Kelvin for Julia-STREAM comparison
    T_outlet_K  = T_outlet_C + 273.15

    print("=" * 60)
    print("Python STREAM reference values (for hardcoding in runtests.jl)")
    print("=" * 60)
    print(f"T_outlet_kelvin = {T_outlet_K:.6f}")
    print(f"mdot            = {mdot:.6f}")
    print()
    print("Hardcode in test/runtests.jl VAL-01 testset:")
    print(f"  T_outlet_ref = {T_outlet_K:.4f}   # K  (Python STREAM: {T_outlet_C:.4f} degC)")
    print(f"  mdot_ref     = {mdot:.6f}  # kg/s")
    print()
    print("Inputs used:")
    print(f"  T_inlet = {T_inlet_C} degC = {T_inlet_K} K")
    print(f"  Q_wall  = {Q_WALL_W} W")
    print(f"  dP_pump = {DP_PUMP} Pa")
    print(f"  n={N_CELLS}, L_ch={L_CH}m, D_h={D_H}m, A_ch={A_CH} m^2")

except ImportError as e:
    print(f"ERROR: Could not import Python STREAM: {e}")
    print(f"Expected path: {STREAM_PATH}")
    print()
    print("If Python STREAM is not available, use these placeholder values")
    print("(replace after running this script with Python STREAM installed):")
    print()
    print("  T_outlet_ref = 0.0   # PLACEHOLDER -- run generate_reference.py")
    print("  mdot_ref     = 0.0   # PLACEHOLDER -- run generate_reference.py")
    sys.exit(1)

except Exception as e:
    print(f"ERROR running Python STREAM: {e}")
    print()
    print("Inspect the Python STREAM API and update the import/call pattern above.")
    print("Key: the script must produce T_outlet in Kelvin and mdot in kg/s.")
    raise
```

```julia
@testset "STREAM Phase 3 Tests" begin

using DifferentialEquations
using Sundials

# ─────────────────────────────────────────────────────────────────
# Reference values from Python STREAM (generate_reference.py)
# Inputs: n=10, L_ch=0.6m, D_h=0.01m, A=7.85e-5 m^2,
#         L_fr=0.3m, dP_pump=3e4 Pa, Q_wall=10kW, T_inlet=313.15 K
# ─────────────────────────────────────────────────────────────────
const T_outlet_ref = 0.0    # REPLACE with value from generate_reference.py (K)
const mdot_ref     = 0.0    # REPLACE with value from generate_reference.py (kg/s)

# ─────────────────────────────────────────────────────────────────
# SYS-01: Closed loop assembles and compiles
# ─────────────────────────────────────────────────────────────────
@testset "SYS-01: Closed loop mtkcompile" begin
    ssys = build_loop()
    @test ssys isa ModelingToolkit.AbstractSystem
end

# ─────────────────────────────────────────────────────────────────
# SYS-02 / SOLV-01: solve_steady returns solution with symbolic indexing
# ─────────────────────────────────────────────────────────────────
@testset "SOLV-01: solve_steady returns solution" begin
    ssys = build_loop()
    n = 10
    T_guess = steady_state_guess(T_inlet=313.15, Q_wall=1e4, mdot_guess=0.1, n=n)
    op = [ssys.ch.T[i] => T_guess[i] for i in 1:n]
    sol = solve_steady(ssys, op)
    # retcode should not be failure
    @test sol.retcode != SciMLBase.ReturnCode.Failure
    # T_outlet should be above inlet (heat is being added)
    T_out = sol[ssys.ch.T_out]
    @test T_out > 313.15
    # T_outlet should be physically bounded (no runaway)
    @test T_out < 450.0
    # mdot should be nonzero
    mdot = abs(sol[ssys.pump.port_in.mdot])
    @test mdot > 1e-4
end

# ─────────────────────────────────────────────────────────────────
# VAL-01: T_outlet and mdot within 1% of Python STREAM reference
# ─────────────────────────────────────────────────────────────────
@testset "VAL-01: Steady-state matches Python STREAM within 1%" begin
    ssys = build_loop()
    n = 10
    T_guess = steady_state_guess(T_inlet=313.15, Q_wall=1e4, mdot_guess=0.1, n=n)
    op = [ssys.ch.T[i] => T_guess[i] for i in 1:n]
    sol = solve_steady(ssys, op)

    T_out = sol[ssys.ch.T_out]
    mdot  = abs(sol[ssys.pump.port_in.mdot])

    @test isapprox(T_out, T_outlet_ref; rtol=0.01)
    @test isapprox(mdot,  mdot_ref;     rtol=0.01)
end

# ─────────────────────────────────────────────────────────────────
# SOLV-02 / VAL-02: Transient -- T_outlet rises after power step
# Qualitative check only: no numerical comparison to Python STREAM
# ─────────────────────────────────────────────────────────────────
@testset "VAL-02: Transient T_outlet rises after power step" begin
    ssys_t, Q_sym = build_loop_transient()
    n = 10
    T_guess = steady_state_guess(T_inlet=313.15, Q_wall=1e4, mdot_guess=0.1, n=n)
    op = [ssys_t.ch.T[i] => T_guess[i] for i in 1:n]

    # Simulate 60s: Q_wall steps from 10kW to 20kW at t=10s
    sol = solve_transient(ssys_t, Q_sym, op, (0.0, 60.0);
                          Q_wall_final = 2.0e4, t_step = 10.0)

    @test sol.retcode != SciMLBase.ReturnCode.Failure
    T_ts = sol[ssys_t.ch.T_out, :]
    @test !any(isnan, T_ts)           # no solver divergence
    @test !any(isinf, T_ts)           # no solver divergence
    # T at end should be above T at start (power doubled, temperature should rise)
    @test T_ts[end] > T_ts[1]
end
```
