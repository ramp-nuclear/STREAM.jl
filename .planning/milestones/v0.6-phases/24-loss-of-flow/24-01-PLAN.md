    **IC Strategy (per D-09, committed):** Use `solve_steady` on the full LOF system with `Pump(0.0)` and `Flapper T_open=1e30`. With `Pump(0.0)` and `g_acc=-9.80665` (downward channel), solve_steady finds the gravity-driven forced-flow equilibrium where friction balances the hydrostatic head `rho*g*L`. KINSOL ignores continuous events, so the Flapper acts as pure `R_closed=1e8` resistance. This produces a consistent (mdot_ss, T[1:n]) state. The Inertia IC is set to this mdot_ss. The transient perturbation comes from buoyancy evolving as the channel heats: as T increases, the density-driven head weakens, friction exceeds drive, and mdot decays through the Inertia time constant until the Flapper fires.

    **Create `test/test_loss_of_flow.jl`:**

    ```julia
    using Test
    using ModelingToolkit
    using ModelingToolkit: t_nounits as t
    using DifferentialEquations
    using STREAM

    @testset "Loss-of-Flow Validation" begin
        n = 10
        T_inlet = 313.15
        T_wall = 373.15
        ssys = STREAM.build_loop_lof(n=n, T_wall=T_wall, T_inlet=T_inlet)

        # ── Steady-state IC extraction (D-09) ──
        op_ss = Pair{Any,Any}[
            [ssys.ch.T[i] => 313.15 + i * 5.0 for i in 1:n]...,
            ssys.ine.inlet.mdot => 0.05,
            ssys.flapper.T_open   => 1e30,
        ]
        sol_ss = solve_steady(ssys, op_ss)
        mdot_ss = sol_ss[ssys.ine.inlet.mdot]
        @test abs(mdot_ss) > 1e-4   # sanity: nonzero forced-flow

        # ── Transient op with ICs from solve_steady ──
        op = Pair{Any,Any}[
            [ssys.ch.T[i] => sol_ss[ssys.ch.T[i]] for i in 1:n]...,
            ssys.ine.inlet.mdot  => mdot_ss,
            ssys.flapper.T_open    => 1e30,
            ssys.flapper.threshold => 0.1 * abs(mdot_ss),  # D-10
        ]
        t_end = 300.0
        t_arr = range(0.0, t_end; length=3000)
        sol = solve_transient(ssys, op, t_arr)
        @test sol.retcode == ReturnCode.Success

        # Flapper opened (T_open < 1e30 sentinel)
        @test sol[ssys.flapper.T_open, end] < 1e10

        # ── VAL-01: energy balance at 5 checkpoints (D-13) ──
        @testset "VAL-01: energy balance throughout transient" begin
            fracs = [0.1, 0.3, 0.5, 0.7, 0.9]
            assertions_run = 0
            for frac in fracs
                i_t = searchsortedfirst(t_arr, frac * t_end)
                mdot_t  = sol[ssys.ine.inlet.mdot, i_t]
                T_1     = sol[ssys.ch.T[1], i_t]
                T_n     = sol[ssys.ch.T[n], i_t]
                cp_t    = cp_water((T_1 + T_n) / 2)
                Q_meas  = abs(mdot_t) * cp_t * abs(T_n - T_1)
                Q_wall_t = sum(sol[ssys.ch.q_wall[j], i_t] for j in 1:n)
                # Skip near-zero checkpoints (reversal transient)
                if abs(Q_wall_t) > 10.0 && abs(mdot_t) > 1e-4
                    @test isapprox(Q_meas, Q_wall_t; rtol=0.05)
                    assertions_run += 1
                end
            end
            @test assertions_run >= 3  # at least 3 of 5 checkpoints must be non-trivial
        end

        # ── VAL-02: NC quasi-steady energy balance (D-12, D-14) ──
        @testset "VAL-02: natural circulation energy balance" begin
            nc_start_idx = searchsortedfirst(t_arr, 0.9 * t_end)
            nc_indices = nc_start_idx:length(t_arr)

            # Average over last 10% window
            mdot_nc_avg = sum(sol[ssys.ine.inlet.mdot, i] for i in nc_indices) / length(nc_indices)
            T1_avg = sum(sol[ssys.ch.T[1], i] for i in nc_indices) / length(nc_indices)
            Tn_avg = sum(sol[ssys.ch.T[n], i] for i in nc_indices) / length(nc_indices)

            # Confirm flow actually reversed (NC established)
            @test mdot_nc_avg < 0 || abs(mdot_nc_avg) < abs(mdot_ss) * 0.1

            cp_avg = cp_water((T1_avg + Tn_avg) / 2)
            Q_meas_nc = abs(mdot_nc_avg) * cp_avg * abs(Tn_avg - T1_avg)

            Q_wall_sum = 0.0
            for i in nc_indices
                Q_wall_sum += sum(sol[ssys.ch.q_wall[j], i] for j in 1:n)
            end
            Q_wall_nc_avg = Q_wall_sum / length(nc_indices)

            @test isapprox(Q_meas_nc, Q_wall_nc_avg; rtol=0.10)
        end
    end
    ```
