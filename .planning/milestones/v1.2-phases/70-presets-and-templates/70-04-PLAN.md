        ```
        let cancelled = false;
        const unwatchers: UnwatchFn[] = [];
        async function setup() {
          const { appConfigDir, join } = await import("@tauri-apps/api/path");
          const { watch, mkdir } = await import("@tauri-apps/plugin-fs");
          const libDir = await join(await appConfigDir(), "presets");
          await mkdir(libDir, { recursive: true }).catch(() => {});
          if (cancelled) return;
          await refreshPresetsDir("library", libDir);
          const unwatchLib = await watch(
            libDir,
            () => { refreshPresetsDir("library", libDir).catch(console.error); },
            { delayMs: 200 }
          );
          if (cancelled) { unwatchLib(); return; }
          unwatchers.push(unwatchLib);

          if (currentProjectDir) {
            const projDir = await join(currentProjectDir, "presets");
            await mkdir(projDir, { recursive: true }).catch(() => {});
            if (cancelled) return;
            await refreshPresetsDir("project", projDir);
            const unwatchProj = await watch(
              projDir,
              () => { refreshPresetsDir("project", projDir).catch(console.error); },
              { delayMs: 200 }
            );
            if (cancelled) { unwatchProj(); return; }
            unwatchers.push(unwatchProj);
          } else {
            // No project open — clear projectPresets via the slice setter
            useStore.getState().setProjectPresets([]);
          }
          setLoading(false);
        }
        setup().catch(err => { console.error("PresetsPanel watcher setup failed", err); setLoading(false); });
        return () => { cancelled = true; unwatchers.forEach(fn => fn()); };
        ```

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| Tauri watch callback → React state | The watcher fires on FS events; if the callback mutates React state after unmount or after the project changes, we leak / corrupt state. |
| External writes to `~/.config/com.stream.composer/presets/` → store | Arbitrary `.scpr` files (even hand-written) land in the slice via the watcher. |
| Reveal in Finder (`revealItemInDir`) → OS shell | Tauri opens the file manager scoped to the file path. |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-70-14 | Tampering | Stale watcher updates after project switch | mitigate | `cancelled` flag inside the `useEffect` setup async function; outer cleanup calls every `unwatchers[i]()`. The deps array `[currentProjectDir]` triggers cleanup before re-setup. |
| T-70-15 | Information Disclosure | `revealItemInDir(filePath)` | mitigate | The opener plugin scopes the reveal action to the file path — it cannot read other directories. The ACL from 70-01 allows `$APPCONFIG/presets/**` and `$APPDATA/...` only. |
| T-70-16 | Tampering | External `.scpr` writes | mitigate | `refreshPresetsDir` (in 70-03) validates via strict `deserializePreset`; bad files are skipped with `console.error`. The user-facing surface remains intact (no crash). |
| T-70-17 | Denial of Service | Watcher fire-storm on a sync app like Dropbox | mitigate | Tauri `delayMs: 200` debounce inside the plugin. Per RESEARCH.md Pitfall 3, this is sufficient for editor save bursts. Sustained high-frequency external writes would still be tolerable since each `refreshPresetsDir` is bounded by directory size. |
| T-70-SC | Tampering | npm installs | n/a | No new npm packages added in this plan. |
</threat_model>
