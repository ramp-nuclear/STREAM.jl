    @testset "ISCB: In-loop SCB Correction" begin
        using ModelingToolkit
        using DifferentialEquations

        geom = PipeGeometry_rectangular(0.625, 0.00127, 0.07, 0.07)
        n_cells = 3

        # Helper: build minimal loop with ChannelAndContacts + Pump + ConstantTemperature
        # Returns (compiled_sys, prob) pair
        function _build_scb_test_loop(; scb_correction=nothing, T_wall_bc=500.0)
            @named ch = ChannelAndContacts(n=n_cells, geometry=geom,
                                           scb_correction=scb_correction)
            @named pump = Pump(1000.0; name=:pump)  # fixed dP
            # Constant temperature BCs on both sides
            @named ht_left = [ConstantTemperature(T_wall_bc; name=Symbol(:ht_left, i)) for i in 1:n_cells]
            @named ht_right = [ConstantTemperature(T_wall_bc; name=Symbol(:ht_right, i)) for i in 1:n_cells]

            # Build connections
            conns = [
                port(pump.outlet, ch.inlet),
                port(ch.outlet, pump.inlet),
            ]
            for i in 1:n_cells
                push!(conns, connect(ht_left[i].port, ch.thermal_left[i]))
                push!(conns, connect(ht_right[i].port, ch.thermal_right[i]))
            end

            sys = compose_systems(:loop, [pump, ch, ht_left..., ht_right...], conns;
                                  guesses = [pump.inlet.P => 1e5])
            csys = mtkcompile(sys)
            prob = ODEProblem(csys, [], (0.0, 100.0))
            return csys, prob
        end

        @testset "ISCB-01: SCB ChannelAndContacts solves" begin
            scb_fn = regime_dependent_q_scb(pressure=1e5)
            csys, prob = _build_scb_test_loop(scb_correction=scb_fn, T_wall_bc=500.0)
            sol = solve(prob, Rodas5P(); abstol=1e-8, reltol=1e-8)
            @test sol.retcode == ReturnCode.Success
        end

        @testset "ISCB-01: Default (no SCB) backward compatibility" begin
            csys_default, prob_default = _build_scb_test_loop(scb_correction=nothing, T_wall_bc=350.0)
            sol_default = solve(prob_default, Rodas5P(); abstol=1e-8, reltol=1e-8)
            @test sol_default.retcode == ReturnCode.Success
        end

        @testset "ISCB-02: High T_wall -> enhanced HTC" begin
            # T_wall = 500K >> T_sat (~373K at 1 atm) -> SCB active
            scb_fn = regime_dependent_q_scb(pressure=1e5)
            csys_scb, prob_scb = _build_scb_test_loop(scb_correction=scb_fn, T_wall_bc=500.0)
            sol_scb = solve(prob_scb, Rodas5P(); abstol=1e-8, reltol=1e-8)

            csys_noscb, prob_noscb = _build_scb_test_loop(scb_correction=nothing, T_wall_bc=500.0)
            sol_noscb = solve(prob_noscb, Rodas5P(); abstol=1e-8, reltol=1e-8)

            # At least one cell should have higher h_tc with SCB
            # Extract h_tc at final time for comparison
            # Use sol[sys.ch.h_tc[i]] pattern
            htc_scb = [sol_scb[csys_scb.ch.h_tc[i]][end] for i in 1:n_cells]
            htc_noscb = [sol_noscb[csys_noscb.ch.h_tc[i]][end] for i in 1:n_cells]
            @test any(htc_scb .> htc_noscb)
        end

        end
    end
