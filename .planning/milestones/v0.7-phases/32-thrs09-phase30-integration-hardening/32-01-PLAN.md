From test/test_channel.jl — minimal ChannelAndContacts loop (use as E2E test template):
```julia
n = 5; T_inlet = 313.15; T_wall = 373.15; L_ch = 0.6; D_ch = 0.01; dP_pump = 3.0e4

@named pump = Pump(dP_pump)
@named cac  = ChannelAndContacts(n=n, geometry=PipeGeometry_circular(L_ch, D_ch))
@named bc   = HeatExchanger(T_inlet)
ct_l = [ConstantTemperature(T_wall; name=Symbol(:ct_l, i)) for i in 1:n]
ct_r = [ConstantTemperature(T_wall; name=Symbol(:ct_r, i)) for i in 1:n]
conns = [
    connect(pump.outlet, bc.inlet),
    connect(bc.outlet, cac.inlet),
    connect(cac.outlet, pump.inlet),
    [connect(ct_l[i].thermal, getproperty(cac, Symbol(:thermal_left,  i))) for i in 1:n]...,
    [connect(ct_r[i].thermal, getproperty(cac, Symbol(:thermal_right, i))) for i in 1:n]...,
    pump.inlet.P ~ 2e5,
]
@named sys = compose(System(conns, t; name=:sys), pump, bc, cac, ct_l..., ct_r...)
ssys = mtkcompile(sys)
op = [ssys.cac.T[i] => T_inlet + (T_wall - T_inlet)*i/n for i in 1:n]
push!(op, ssys.cac.inlet.mdot => 0.490)
sol = solve_steady(ssys, op)
```

From src/physical_models/htc/correlations.jl:
```julia
function fully_developed_laminar_h_spl(; Dh, aspect_ratio)
    nu = _two_sided_heating_nusselt(aspect_ratio)
    return (Re, Pr, args...) -> nu
end

function developing_laminar_h_spl(; Dh, develop_length, aspect_ratio)
    correction = 6 - 5 * exp(-0.75 * aspect_ratio / 0.3257)
    return (Re, Pr, args...) -> begin
        x_star = develop_length / Dh / Re / Pr / correction
        nudev = _nusselt_coefficient_developing(x_star)
        _two_sided_heating_nusselt(aspect_ratio, nudev)
    end
end
```
Note: developing_laminar_h_spl calls _nusselt_coefficient_developing(x_star) which has
`if x <= 2e-4` — this is a plain Julia if/else on a symbolic Num expression. This will
collapse the branch at trace time, which is fine for the factory pattern since x_star
is evaluated symbolically only once per cell. If it causes a tracing error, the fix is
to restructure using ifelse() — but first verify whether it actually errors.
