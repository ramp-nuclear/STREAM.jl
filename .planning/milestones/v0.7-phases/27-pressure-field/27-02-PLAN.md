function sat_temperature(P_Pa::Real)
    X = log(abs(P_Pa) * 1e-6)
    A = 179.9600321; B = -0.1063030; C = 24.2278298; D = 2.951e-4
    T_C = (A + C * X) / (1 + B * X + D * X^2)
    return T_C + 273.15
end
@register_symbolic sat_temperature(P::Real)
