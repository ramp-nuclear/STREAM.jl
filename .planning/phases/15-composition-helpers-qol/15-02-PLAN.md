```julia
# helpers.jl — Composition helpers for MTR subsystem assembly (COMP-01/02/03/04)
#
# Design principle: helpers take pre-built component instances and wire them.
# Callers are responsible for matching dimensions (n == nz) and naming.
# All helpers return a raw ODESystem via compose() — caller calls mtkcompile().
# build_initializeprob=false must be used when solving HeatDiffusion+CAC systems.

using ModelingToolkit
using ModelingToolkit: t_nounits as t

# ----------------------------------------------------------------
# symmetric_plate — COMP-01
# Wires one HeatDiffusion fuel plate to one ChannelAndContacts:
#   cac.thermal_right[i] <-> fuel.thermal_left[i]   (left face of plate = right face of channel)
#   cac.thermal_left[i]  <-> fuel.thermal_right[i]  (right face of plate = left face of channel)
#
# "Symmetric" means both plate faces heat the same channel — the plate
# is sandwiched symmetrically. cac.n must equal fuel.nz.
#
# Returns raw ODESystem. Caller must add boundary conditions (Pump, pressure
# anchor, inlet T) and call mtkcompile().
# ----------------------------------------------------------------
function symmetric_plate(cac, fuel; name::Symbol)
    n = length(ModelingToolkit.get_systems(cac))  # fallback
    # Determine n from ChannelAndContacts by inspecting thermal port count
    # Use the port() helper pattern to get port list length via try/catch
    # Conservative: infer n from the cac system's subsystems named thermal_leftN
    all_sub_names = Symbol.(nameof.(ModelingToolkit.get_systems(cac)))
    n = count(s -> startswith(string(s), "thermal_left"), all_sub_names)

    connections = Equation[]
    for i in 1:n
        push!(connections, connect(port(cac, :thermal_right, i), port(fuel, :thermal_left, i)))
        push!(connections, connect(port(cac, :thermal_left,  i), port(fuel, :thermal_right, i)))
    end

    @named inner = ODESystem(connections, t; systems=[cac, fuel])
    return rename(inner, name)
end

# ----------------------------------------------------------------
# plate — COMP-02
# Wires one HeatDiffusion plate between two independent channels:
#   ch_left.thermal_right[i]  <-> fuel.thermal_left[i]
#   ch_right.thermal_left[i]  <-> fuel.thermal_right[i]
#
# ch_left.n == ch_right.n == fuel.nz required (caller responsibility).
# ----------------------------------------------------------------
function plate(ch_left, ch_right, fuel; name::Symbol)
    all_sub_names = Symbol.(nameof.(ModelingToolkit.get_systems(ch_left)))
    n = count(s -> startswith(string(s), "thermal_left"), all_sub_names)

    connections = Equation[]
    for i in 1:n
        push!(connections, connect(port(ch_left,  :thermal_right, i), port(fuel, :thermal_left, i)))
        push!(connections, connect(port(ch_right, :thermal_left,  i), port(fuel, :thermal_right, i)))
    end

    @named inner = ODESystem(connections, t; systems=[ch_left, ch_right, fuel])
    return rename(inner, name)
end

# ----------------------------------------------------------------
# one_sided_connection — COMP-03
# Wires one HeatDiffusion plate to one ChannelAndContacts on one side only.
# side=:left  => cac.thermal_left[i]  <-> fuel.thermal_right[i]
# side=:right => cac.thermal_right[i] <-> fuel.thermal_left[i]
# The opposite fuel face remains unconnected (adiabatic by MTK default).
# ----------------------------------------------------------------
function one_sided_connection(channel, fuel; side::Symbol = :left, name::Symbol)
    @assert side in (:left, :right) "side must be :left or :right, got :$side"

    all_sub_names = Symbol.(nameof.(ModelingToolkit.get_systems(channel)))
    n = count(s -> startswith(string(s), "thermal_left"), all_sub_names)

    connections = Equation[]
    if side == :left
        for i in 1:n
            push!(connections, connect(port(channel, :thermal_left, i), port(fuel, :thermal_right, i)))
        end
    else
        for i in 1:n
            push!(connections, connect(port(channel, :thermal_right, i), port(fuel, :thermal_left, i)))
        end
    end

    @named inner = ODESystem(connections, t; systems=[channel, fuel])
    return rename(inner, name)
end

# ----------------------------------------------------------------
# compose_systems — COMP-04
# Thin wrapper: merges two or more independently-built ODESystems with
# explicit cross-connections (a Vector{Equation} of connect() calls).
#
# Usage:
#   connections = [connect(plate1.cac.port_out, plate2.cac.port_in), ...]
#   top = compose_systems(plate1, plate2, connections; name=:reactor)
#
# Equivalent to: ODESystem(connections, t; systems=[sys_a, sys_b, ...], name=name)
# Does NOT call mtkcompile — caller does that.
# ----------------------------------------------------------------
function compose_systems(systems::ModelingToolkit.AbstractSystem...; connections::Vector{<:Equation}, name::Symbol)
    ODESystem(connections, t; systems=collect(systems), name=name)
end
```

NOTE on n-detection: The pattern `count(s -> startswith(string(s), "thermal_left"), all_sub_names)` uses ModelingToolkit.get_systems() to find subsystems named thermal_left1, thermal_left2... This is the same subsystem structure created by the ChannelAndContacts constructor. If this pattern is fragile, fall back to a keyword argument: add `n::Int` as an optional kwarg to each helper with a default of `nothing` and detect when absent.
