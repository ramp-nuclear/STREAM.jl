```julia
# ----------------------------------------------------------------
# port — QOL-03
# Wraps getproperty(sys, Symbol(port_name, i)) for MTK port array access.
# Avoids the verbose Symbol concatenation pattern at call sites.
# Example: port(sys, :thermal_left, 2) == getproperty(sys, :thermal_left2)
# ----------------------------------------------------------------
function port(sys::ModelingToolkit.AbstractSystem, port_name::Symbol, i::Int)
    getproperty(sys, Symbol(port_name, i))
end

# ----------------------------------------------------------------
# check_gravity_mismatch — QOL-02
# Checks whether gravity pressure terms sum to zero at zero flow.
# Algorithm: substitute mdot=0 into all pressure equations from the
# compiled system, check whether the resulting system is self-consistent.
#
# Returns :ok if the pressure system is consistent at zero flow.
# Returns :mismatch if gravity terms produce a non-zero pressure residual,
# with a @warn showing the residual magnitude.
#
# Accepts pre- or post-mtkcompile ODESystem.
# ----------------------------------------------------------------
function check_gravity_mismatch(sys::ModelingToolkit.AbstractSystem)
    # Use ModelingToolkit to extract and substitute symbolically
    eqs = equations(sys)
    # Collect all mdot unknowns (port_in.mdot, port_out.mdot appear in pressure eqs)
    all_vars = unknowns(sys)
    mdot_vars = filter(v -> occursin("mdot", string(v)), all_vars)

    # Build substitution dict: all mdot variables -> 0
    subs = Dict(v => 0.0 for v in mdot_vars)

    # Extract pressure difference equations (dP equations and port pressure equations)
    pressure_eqs = filter(eq -> occursin("dP", string(eq)) || occursin(".P", string(eq)), eqs)

    if isempty(pressure_eqs)
        @warn "check_gravity_mismatch: no pressure equations found in system"
        return :ok
    end

    # Substitute mdot=0 and evaluate residuals
    residuals = Float64[]
    for eq in pressure_eqs
        lhs = substitute(eq.lhs - eq.rhs, subs)
        try
            val = Float64(lhs)
            push!(residuals, abs(val))
        catch
            # symbolic remainder — skip (may still have T unknowns)
        end

    end

    if isempty(residuals)
        # Could not evaluate numerically (T unknowns remain) — skip check
        return :ok
    end

    max_residual = maximum(residuals)
    threshold = 1.0  # 1 Pa threshold for gravity mismatch
    if max_residual > threshold
        @warn "check_gravity_mismatch: gravity pressure residual at zero flow = $(round(max_residual; digits=2)) Pa — loop gravity terms are unbalanced"
        return :mismatch
    end
    return :ok
end
```
