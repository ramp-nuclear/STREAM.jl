    Modify gui/src/App.tsx:
    1. Import `AutoRecoverRestoreModal`, `detectCrashOnLaunch` from `@/lib/autoRecover`, `initAutoRecover` from `@/store/useStore`, `invoke` from `@tauri-apps/api/core`.
    2. Add state: `const [restoreCandidates, setRestoreCandidates] = useState<RestoreCandidate[] | null>(null);` — null = not yet checked, [] = checked-no-crash, [...candidates] = crash detected.
    3. Add a ref `const teardownRef = useRef<(() => Promise<void>) | null>(null);`.
    4. Mount effect:
       ```tsx
       useEffect(() => {
         let canceled = false;
         (async () => {
           const pid = await invoke<number>('get_pid');
           const result = await detectCrashOnLaunch(pid);
           if (canceled) return;
           if (result.crashed) {
             const candidates = result.sidecars.map((basename) => buildRestoreCandidate(basename, result.staleLockfile));
             setRestoreCandidates(candidates);
           } else {
             setRestoreCandidates([]);
             const { teardown } = await initAutoRecover();
             teardownRef.current = teardown;
           }
         })();
         return () => {
           canceled = true;
           if (teardownRef.current) {
             teardownRef.current();
             teardownRef.current = null;
           }
         };
       }, []);
       ```
       Where `buildRestoreCandidate(basename, staleLockfile)` is a small helper:
       - basename starting with 'untitled-' → displayName = 'Unsaved project'; else displayName = basename.replace(/\.scp\.autosave$/, '').
       - modifiedAt = staleLockfile?.startedAt ?? new Date().toISOString().
    5. Modal resolution handlers (W12 — try/finally so initAutoRecover always runs):
       ```tsx
       async function handleRecover(basename: string) {
         try {
           await useStore.getState().recoverFromSidecar(basename);
         } catch (err) {
           console.error('[AutoRecover] Recover failed:', err);
           // Continue to initAutoRecover regardless — the user has dismissed the modal
           // by clicking Recover, and the next session needs its own lockfile + writer.
         } finally {
           try {
             const { teardown } = await initAutoRecover();
             teardownRef.current = teardown;
           } catch (err) {
             console.error('[AutoRecover] initAutoRecover after Recover failed:', err);
           }
           setRestoreCandidates([]);
         }
       }
       async function handleDiscard() {
         try {
           await useStore.getState().discardAllSidecars();
         } catch (err) {
           console.error('[AutoRecover] Discard failed:', err);
         } finally {
           try {
             const { teardown } = await initAutoRecover();
             teardownRef.current = teardown;
           } catch (err) {
             console.error('[AutoRecover] initAutoRecover after Discard failed:', err);
           }
           setRestoreCandidates([]);
         }
       }
       ```
       Rationale: `recoverFromSidecar` parses untrusted JSON via `deserializeProject` — a malformed sidecar can throw. The user has already committed to "Recover" by clicking the button; we must not strand them with a modal still visible AND no autoRecover writer running. `try/finally` guarantees the modal closes (`setRestoreCandidates([])`) AND the new session's writer/lockfile come online, even if hydration partially failed.
    6. Render gate:
       ```tsx
       if (restoreCandidates === null) return <BootSplash />;     // brief — could be just an empty <div>
       if (restoreCandidates.length > 0) {
         return <AutoRecoverRestoreModal candidates={restoreCandidates} onRecover={handleRecover} onDiscard={handleDiscard} />;
       }
       // ... existing App body (canvas + sidebars) follows
       ```
