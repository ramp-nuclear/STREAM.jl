    Update gui/src/App.tsx to add three useEffect hooks:

    **A. Keyboard shortcuts useEffect** (global keydown listener):

    ```typescript
    useEffect(() => {
      const handleKeyDown = async (e: KeyboardEvent) => {
        // Ctrl+S or Cmd+S — Save
        if ((e.ctrlKey || e.metaKey) && e.key === "s" && !e.shiftKey) {
          e.preventDefault();
          await useStore.getState().saveProject();
        }
        // Ctrl+Shift+S or Cmd+Shift+S — Save As
        if ((e.ctrlKey || e.metaKey) && e.key === "s" && e.shiftKey) {
          // Note: capital S check not needed, shiftKey flag is sufficient
          e.preventDefault();
          await useStore.getState().saveProjectAs();
        }
        // Ctrl+O or Cmd+O — Open
        if ((e.ctrlKey || e.metaKey) && e.key === "o") {
          e.preventDefault();
          const { isDirty } = useStore.getState();
          if (isDirty) {
            const action = await promptUnsavedChanges();
            if (action === "cancel") return;
            if (action === "save") await useStore.getState().saveProject();
          }
          await useStore.getState().loadProject();
        }
        // Ctrl+N or Cmd+N — New
        if ((e.ctrlKey || e.metaKey) && e.key === "n") {
          e.preventDefault();
          const { isDirty } = useStore.getState();
          if (isDirty) {
            const action = await promptUnsavedChanges();
            if (action === "cancel") return;
            if (action === "save") await useStore.getState().saveProject();
          }
          await useStore.getState().newProject();
        }
      };
      window.addEventListener("keydown", handleKeyDown);
      return () => window.removeEventListener("keydown", handleKeyDown);
    }, []);
    ```

    Import `promptUnsavedChanges` from `./components/FileMenu`.

    **C. Window close guard useEffect** (per D-10, Pitfall 3):

    ```typescript
    useEffect(() => {
      let unlisten: (() => void) | undefined;
      getCurrentWindow().onCloseRequested(async (event) => {
        const { isDirty } = useStore.getState();
        if (!isDirty) return; // Allow close

        event.preventDefault(); // MUST call synchronously before async work
        const action = await promptUnsavedChanges();
        if (action === "save") {
          await useStore.getState().saveProject();
          await getCurrentWindow().close();
        } else if (action === "discard") {
          // Force close without isDirty re-check:
          // Temporarily clear isDirty so the recursive close doesn't re-trigger
          useStore.setState({ isDirty: false });
          await getCurrentWindow().close();
        }
        // "cancel" -> do nothing, window stays open
      }).then((fn) => { unlisten = fn; });

      return () => { unlisten?.(); };
    }, []);
    ```
