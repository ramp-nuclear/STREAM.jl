```julia
function solve_steady(ssys, op; abstol=1e-8, reltol=1e-6)
    prob = SteadyStateProblem(ssys, op; warn_initialize_determined=false)
    sol  = solve(prob, SSRootfind(KINSOL()); abstol=abstol, reltol=reltol)
    return sol
end
```

```julia
function build_loop(; ...)
    @named pump = Pump(dP_pump=dP_pump)
    @named ch   = Channel(...)
    @named bc   = _make_temp_bc(...)
    connections = [
        connect(pump.port_out, bc.port_in),
        ...
        pump.port_in.P ~ 1.0e5,   # pressure anchor
    ]
    @named sys = compose(System(connections, t; name=:sys), pump, bc, ch)
    ssys = mtkcompile(sys)
    return ssys
end
```

```julia
# ----------------------------------------------------------------
# build_cube
# Assembles the classic Resistor Cube network:
#   12 Resistors on cube edges, 8 corner nodes, 1 Pump driving
#   body-diagonal flow (corner 0 = source, corner 7 = sink).
#
# Analytical equivalent resistance (body diagonal): 5/6 * R
# Analytical total flow: mdot = dP_pump / (5/6 * R)
#
# Corner labeling: binary xyz bits (0=000, 7=111)
# Edges per axis:
#   X (bit-2): r04, r15, r26, r37
#   Y (bit-1): r02, r13, r46, r57
#   Z (bit-0): r01, r23, r45, r67
#
# Junction handling: MTK multi-port connect(a, b, c) generates
# Kirchhoff equations automatically — no Junction component needed.
#
# Returns compiled ssys. Access pump.port_out.mdot for total flow.
# ----------------------------------------------------------------
function build_cube(; dP_pump=3.0e4, R=1.0e4)
    @named pump = Pump(dP_pump=dP_pump)
    # 12 resistors — one per cube edge
    @named r01 = Resistor(R=R); @named r02 = Resistor(R=R); @named r04 = Resistor(R=R)
    @named r13 = Resistor(R=R); @named r15 = Resistor(R=R)
    @named r23 = Resistor(R=R); @named r26 = Resistor(R=R)
    @named r37 = Resistor(R=R)
    @named r45 = Resistor(R=R); @named r46 = Resistor(R=R)
    @named r57 = Resistor(R=R); @named r67 = Resistor(R=R)

    connections = [
        # Corner 0 (source): pump.port_out + 3 resistor inlets
        connect(pump.port_out, r01.port_in, r02.port_in, r04.port_in),
        # Corner 1: r01.port_out + r13.port_in + r15.port_in
        connect(r01.port_out, r13.port_in, r15.port_in),
        # Corner 2: r02.port_out + r23.port_in + r26.port_in
        connect(r02.port_out, r23.port_in, r26.port_in),
        # Corner 3: r13.port_out + r23.port_out + r37.port_in
        connect(r13.port_out, r23.port_out, r37.port_in),
        # Corner 4: r04.port_out + r45.port_in + r46.port_in
        connect(r04.port_out, r45.port_in, r46.port_in),
        # Corner 5: r15.port_out + r45.port_out + r57.port_in
        connect(r15.port_out, r45.port_out, r57.port_in),
        # Corner 6: r26.port_out + r46.port_out + r67.port_in
        connect(r26.port_out, r46.port_out, r67.port_in),
        # Corner 7 (sink): pump.port_in + 3 resistor outlets
        connect(pump.port_in, r37.port_out, r57.port_out, r67.port_out),
        # Boundary conditions
        pump.port_in.P ~ 1.0e5,    # pressure gauge freedom fix
    ]

    @named sys = compose(
        System(connections, t; name=:sys),
        pump, r01, r02, r04, r13, r15, r23, r26, r37, r45, r46, r57, r67
    )

    t_compile = @elapsed ssys = mtkcompile(sys)
    n_eq = length(equations(ssys))
    n_uk = length(unknowns(ssys))
    @info "build_cube compile time: $(round(t_compile; digits=2))s" n_equations=n_eq n_unknowns=n_uk

    return ssys
end
```
