const enrichedNodes = useMemo(() => {
  if (activeLayer === "Both") return nodes;
  return nodes.map(node => {

**Step 2: Update CanvasPanel.tsx — Tab key interception (D-11)**

Add an `onKeyDown` handler to the container div. The Tab handler cycles layers only when the event target is NOT a form element:

```typescript
const handleContainerKeyDown = useCallback((e: React.KeyboardEvent) => {
  if (e.key !== "Tab") return;
  // Don't intercept Tab inside form elements
  const target = e.target as HTMLElement;
  if (
    target instanceof HTMLInputElement ||
    target instanceof HTMLTextAreaElement ||
    target instanceof HTMLSelectElement ||
    target.isContentEditable
  ) {
    return;
  }
  e.preventDefault();
  e.stopPropagation();
  cycleLayer();
}, [cycleLayer]);
```

Add `onKeyDown={handleContainerKeyDown}` to the container div:
```tsx
<div ref={containerRef} className="flex-1 h-full relative" tabIndex={-1} onKeyDown={handleContainerKeyDown}>
```
