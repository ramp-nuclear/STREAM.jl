        // Pre-paint auto-extend preview here (so the highlight is visible while the modal animates in)

        // Lazy-import autoExtendSelection to avoid pulling presetIO into FileMenu's module graph eagerly
        import("../lib/presetIO").then(({ autoExtendSelection }) => {
          const { extendedIds } = autoExtendSelection(selectedIds, nodes, edges);
          const extras = new Set([...extendedIds].filter(id => !selectedIds.has(id)));
          if (extras.size > 0) {
            useStore.setState(state => ({
              nodes: state.nodes.map(n =>
                extras.has(n.id) ? { ...n, data: { ...n.data, autoExtended: true } } : n

        });

      const handleLoadPreset = async () => {

          await useStore.getState().loadPresetFromPath({ x: centerX, y: centerY });

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| Custom DOM event `stream:open-save-preset` → modal open state | Any code on the page can dispatch this. |
| Canvas drop `dataTransfer` → store action | The drag payload is a JSON string read from `dataTransfer`; in Tauri WebView it is sourced from the same-origin app, but defense in depth still parses safely. |
| File picker → `loadPresetAtPosition` | The user selects an arbitrary `.scpr` from disk. |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-70-22 | Spoofing | Custom-event `stream:open-save-preset` | accept | Tauri WebView is same-origin and locally trusted; no third-party scripts run. The modal is harmless on its own — saving requires explicit user clicks. |
| T-70-23 | Tampering | dataTransfer `application/stream-preset` payload | mitigate | The drop handler `JSON.parse`s inside a try/catch. The result is passed to `loadPresetAtPosition`, which reads the actual file via Tauri FS and calls `deserializePreset` (strict version/kind check). A corrupted payload that names a non-existent path throws cleanly. |
| T-70-24 | Tampering | User-selected `.scpr` from file picker | mitigate | `deserializePreset` enforces `format_version` and `kind`. Loading a malicious `.scpr` cannot execute code; it can at worst add visually corrupt nodes, which Phase 71 validation will surface. |
| T-70-25 | Information Disclosure | Reveal-in-Finder triggered via opener | accept | Scope already bounded in plan 70-01 by the ACL. |
| T-70-SC | Tampering | No new npm/cargo installs in this plan | n/a | All packages installed in plans 70-01/02. |
