      ```
      export function useValidationFieldHighlight(selectedNodeId: string | null, containerRef: React.RefObject<HTMLElement>) {
        const validationResults = useStore(s => s.validationResults);
        useEffect(() => {
          if (!selectedNodeId || !containerRef.current) return;
          // Remove prior highlights
          containerRef.current.querySelectorAll('[data-field-path]').forEach(el => {
            el.classList.remove('validation-field-error', 'validation-field-warning');
          });
          // Find this-node field targets
          for (const result of validationResults) {
            for (const target of result.targets) {
              if (target.kind !== 'field' || target.nodeId !== selectedNodeId) continue;
              const el = containerRef.current.querySelector(`[data-field-path="${CSS.escape(target.fieldPath)}"]`);
              if (!el) continue;
              const cls = result.severity === 'error' ? 'validation-field-error' : result.severity === 'warning' ? 'validation-field-warning' : null;
              if (cls) el.classList.add(cls);
            }
          }
          // Cleanup on dep change handled by the next effect run (remove all + re-add)
        }, [selectedNodeId, validationResults]);
      }
      ```
